import 'dart:io';

import 'package:flutter/material.dart';
import 'package:flutter_code_editor/flutter_code_editor.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';

Finder _status(String id, String text) => find.descendant(
  of: find.byKey(Key('exercise-$id')),
  matching: find.text(text),
);

Future<void> _waitFor(WidgetTester t, Finder f) async {
  for (var i = 0; i < 60 && f.evaluate().isEmpty; i++) {
    await t.runAsync(() => Future<void>.delayed(const Duration(milliseconds: 500)));
    await t.pump();
  }
}

void main() {
  testWidgets('E2E thật: Home → split-chunks → lỗi cú pháp ô B → sửa → passed → Bài tiếp theo', (t) async {
    HttpOverrides.global = null; // cho HttpClient thật
    final clock = DateTime(2024, 1, 1, 9);
    await t.binding.setSurfaceSize(const Size(420, 900));
    addTearDown(() => t.binding.setSurfaceSize(null));
    appRouter.go('/');
    await t.pumpWidget(ProviderScope(
      overrides: [clockProvider.overrideWithValue(() => clock)],
      child: const SmartCodingStudyApp(),
    ));
    await t.pumpAndSettle();

    await t.tap(find.byKey(const Key('continue-card')));
    await t.pumpAndSettle();
    final ex = exerciseById('split-chunks');
    final answers = {for (final b in ex.blocks) b.label: b.acceptedAnswers.first};

    // 1) Ô B thiếu ngoặc đóng → backend báo syntax ở dòng của ô B (INV-05 xuyên Python↔Dart).
    for (final e in answers.entries) {
      final code = e.key == 'B' ? 'chunks.append(text[i:i + size]' : e.value;
      await t.enterText(find.byKey(Key('cell-${e.key}-field')), code);
      await t.pump();
    }
    await t.tap(find.byKey(const Key('run-button')));
    await _waitFor(t, find.textContaining('Lỗi ở ô'));
    final errText = find.textContaining('Lỗi ở ô').evaluate().map((e) => (e.widget as Text).data).toList();
    // ignore: avoid_print
    print('E2E step1 error texts: $errText');
    expect(find.textContaining('Lỗi ở ô B'), findsWidgets);
    expect(find.byKey(const Key('next-exercise')), findsNothing);

    // 2) Sửa ô B → passed thật.
    await t.enterText(find.byKey(const Key('cell-B-field')), answers['B']!);
    await t.pump();
    await t.tap(find.byKey(const Key('run-button')));
    await _waitFor(t, find.byKey(const Key('next-exercise')));
    expect(find.byKey(const Key('next-exercise')), findsOneWidget);
    // ignore: avoid_print
    print('E2E step2 passed badge: ${find.textContaining('test đạt').evaluate().map((e) => (e.widget as Text).data).toList()}');

    // 3) Bài tiếp theo → overlap-chunks, state KHÔNG bị tái dùng từ split-chunks.
    await t.tap(find.byKey(const Key('next-exercise')));
    await t.pumpAndSettle();
    final overlap = exerciseById('overlap-chunks');
    expect(find.text(overlap.title), findsWidgets);
    await t.tap(find.text('Code 0/${overlap.blocks.length}'));
    await t.pumpAndSettle();
    final fieldA = t.widget<CodeField>(find.byKey(const Key('cell-A-field')));
    // ignore: avoid_print
    print('E2E step3 overlap cell A text: "${fieldA.controller.text}"');
    expect(fieldA.controller.text, isEmpty);

    // 4) Bất biến cũ: Topic Chunking hiện split-chunks Đã xong, overlap-chunks mở.
    await t.tap(find.byKey(const Key('exercise-back')));
    await t.pumpAndSettle();
    await t.tap(find.byKey(const Key('theme-rag')));
    await t.pumpAndSettle();
    await t.tap(find.byKey(const Key('topic-rag-chunking')));
    await t.pumpAndSettle();
    expect(_status('split-chunks', 'Đã xong'), findsOneWidget);
    expect(_status('overlap-chunks', 'Khoá'), findsNothing);
    // ignore: avoid_print
    print('E2E step4 topic: split=${_status('split-chunks', 'Đã xong').evaluate().length} overlapMới=${_status('overlap-chunks', 'Mới').evaluate().length}');

    await t.pumpWidget(const SizedBox());
    await t.pump(const Duration(seconds: 30));
  });
}
